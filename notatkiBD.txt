CREATE OR REPLACE VIEW umiejętności_możliwe_do_opanowania AS
-- przypadek 1: Umiejętność nie jest podrzędna względem żadnej innej 
SELECT 
    p.ID_postaci,
    u.Nazwa AS Nazwa_Umiejętności
FROM Postacie p
    CROSS JOIN Umiejętności u
--- warunek 1.1: Umiejętność nie jest podrzędna względem żadnej innej
WHERE NOT EXISTS (
    SELECT 1 
    FROM Podległość pod 
    WHERE pod.podrzędna = u.Nazwa --indeks?
)
--- Warunek 1.2: Ta konkretna postać jeszcze jej nie opanowała
AND NOT EXISTS (
    SELECT 1

    WHERE o.Postacie_ID = p.ID_postaci 
      AND o.Umiejętności_Nazwa = u.Nazwa
)
UNION
--- przypadek 2: Umiejętności podrzędne, których wymagania (nadrzędne) zostały spełnione przez postać, ale sama podrzędna nie jest jeszcze opanowana
SELECT 
    p.ID_postaci,
    u.Nazwa AS Nazwa_Umiejętności
FROM Postacie p
CROSS JOIN Podległość pod
--- Dołączamy informacje o tym, czy i na jakim poziomie postać opanowała daną umiejętność nadrzędną
LEFT JOIN Opanowane o ON o.Postacie_ID = p.ID_postaci 
                     AND o.Umiejętności_Nazwa = pod.wymagana_nadrzędna 
                     AND o.poziom >= pod.minimalny_poziom
--- Grupowanie pozwala nam zliczyć wymagania dla konkretnej pary (Postać, Umiejętność podrzędna)
GROUP BY p.ID_postaci, u.Nazwa
HAVING 
    --- Warunek 2.1
    ---- Liczba wszystkich wymaganych nadrzędnych dla tej umiejętności
    COUNT(pod.wymagana_nadrzędna) = 
    ---- Liczba spełnionych wymagań przez tę postać (COUNT ignoruje wartości NULL z LEFT JOIN)
    COUNT(o.Umiejętności_Nazwa)
    
    --- Warunek 2.2 końcowy: Postać nie opanowała jeszcze tej podrzędnej umiejętności

CREATE OR REPLACE VIEW umiejętności_możliwe_do_podwyższenia AS
SELECT 
    P.ID_postaci,
    O.Umiejętności_Nazwa AS Nazwa_Umiejętności
FROM Opanowane NadO RIGHT JOIN Podległość Pod ON NadO.Umiejętności_Nazwa=Podległość.wymagana_nadrzędna 
    INNER JOIN Opanowane O ON Pod.podrzędna=O.Umiejętności_Nazwa 
    INNER JOIN Postacie P ON O.Postacie_ID = P.ID_postaci
WHERE O.poziom<COALESCE(NadO.poziom-NadO.minimalny_poziom,P.poziom);

----maksymalne poziomy posiadanych umiejętności dla postaci
--błąd - nie wzięto pod uwagę umiejętności nadrzędnej
CREATE OR REPLACE VIEW maksymalne_poziomy_posiadanych_umiejętności AS
SELECT 
    P.ID_postaci,
    O.Umiejętności_Nazwa,
    O.poziom AS Obecny_poziom
    COALESCE(NadO.poziom-NadO.minimalny_poziom,P.poziom) AS Maksymalny_poziom 
FROM Opanowane NadO RIGHT JOIN Podległość Pod ON NadO.Umiejętności_Nazwa=Podległość.wymagana_nadrzędna 
    INNER JOIN Opanowane O ON Pod.podrzędna=O.Umiejętności_Nazwa 
    INNER JOIN Postacie P ON O.Postacie_ID = P.ID_postaci
WHERE O.poziom<COALESCE(NadO.poziom-NadO.minimalny_poziom,P.poziom);
