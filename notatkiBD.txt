CREATE OR REPLACE FUNCTION oblicz_punkty_klasowe_z_cech
(
    p_postac integer,
    p_klasa varchar
)
RETURNS integer
LANGUAGE plpgsql
AS
$$
DECLARE
    punkty_klasowe_wymagane_do_awansu integer;
BEGIN
    SELECT
       ilość_punktów_klasowych+ilość_punktów_klasowych_z_cech 
    INTO
        punkty_klasowe_wymagane_do_awansu
    FROM Postacie p
    JOIN Klasa_postaci kp
        ON kp.Postacie_ID=p.ID_postaci
        AND kp.Klasa_nazwa=p_klasa
    JOIN Klasy k
        ON k.Nazwa=kp.Klasa_nazwa
    WHERE p.ID_postaci=p_postac;
    RETURN;
END;
$$;

SELECT
    routine_schema AS schemat,
    routine_name AS nazwa_funkcji,
    routine_type AS typ, -- FUNCTION, PROCEDURE
    specific_name AS unikalna_nazwa_kodu
FROM information_schema.routines
WHERE routine_schema = 'public' -- Zmień 'public' na nazwę swojego schematu lub usuń filtr
ORDER BY routine_schema, routine_name;


szczegóły dotyczące funkcji
SELECT
    p.oid::regprocedure AS funkcja,
    p.prosrc AS kod_zrodlowy, -- Kod SQL/PL/pgSQL
    p.prolang AS nazwa_jezzyka, -- np. 'sql', 'plpgsql'
    p.prorettype::regtype AS typ_zwrotny,
    p.proargnames AS nazwy_argumentow
FROM pg_catalog.pg_proc p
JOIN pg_catalog.pg_namespace n ON p.pronamespace = n.oid
WHERE n.nspname = 'public' -- Ograniczenie do schematu public
ORDER BY p.proname;


triggery:
SELECT
    t.tgname AS nazwa_trigera,
    c.relname AS tabela, -- Nazwa tabeli
    n.nspname AS schemat,
    pg_get_expr(t.tgqual, t.tgrelid) AS warunek_when, -- Jeśli istnieje warunek "WHEN"
    p.proname AS funkcja_wywoływana,
    CASE 
        WHEN t.tgtype & 1 THEN 'BEFORE'
        WHEN t.tgtype & 2 THEN 'AFTER'
        WHEN t.tgtype & 4 THEN 'INSTEAD OF'
    END AS czas_wystapienia,
    CASE 
        WHEN t.tgtype & 8 THEN 'ROW'
        WHEN t.tgtype & 16 THEN 'STATEMENT'
    END AS poziom_trigera
FROM pg_catalog.pg_trigger t
JOIN pg_catalog.pg_class c ON t.tgrelid = c.oid
JOIN pg_catalog.pg_namespace n ON c.relnamespace = n.oid
LEFT JOIN pg_catalog.pg_proc p ON t.tgfoid = p.oid
WHERE c.relkind = 'r' -- Tylko zwykłe tabele
ORDER BY c.relname, t.tgname;

indeksy:
SELECT
    schemaname AS schemat,
    relname AS tabela,
    indexname AS nazwa_indeksu,
    idx_scan AS liczba_skanowan,
    idx_tup_read AS przeczytanych_krotek,
    idx_tup_fetch => fetched_krotek
FROM pg_stat_user_indexes
ORDER BY idx_scan DESC;

indeksy szczegóły
SELECT
    n.nspname AS schemat,
    c.relname AS nazwa_tabeli,
    i.relname AS nazwa_indeksu,
    pg_get_indexdef(i.oid) AS definicja_indeksu, -- np. CREATE INDEX ... ON ...
    idx.indisunique AS unikalny,
    idx.indisprimary AS klucz_glowny,
    pg_get_expr(idx.indpred, idx.indrelid) AS warunek_partialny -- Jeśli to indeks częściowy
FROM pg_catalog.pg_index idx
JOIN pg_catalog.pg_class c ON idx.indrelid = c.oid
JOIN pg_catalog.pg_namespace n ON c.relnamespace = n.oid
LEFT JOIN pg_catalog.pg_class i ON i.oid = idx.indexrelid
WHERE c.relkind = 'r' -- Tylko tabele
  AND i.relkind = 'i' -- Tylko indeksy
ORDER BY schemat, nazwa_tabeli, nazwa_indeksu;
